# Program to check leap year

year = int(input("Enter year: "))

    print("Result: Leap Year")

    print("Result: Not a Leap Year")

=========================================
MYSQL – NUMERIC FUNCTIONS
=========================================

-- ROUND()
SELECT product_name,
ROUND(price,2) AS Rounded_Price
FROM products;

-- FLOOR()
SELECT product_name,
FLOOR(price) AS Floor_Value
FROM products;

-- CEIL()
SELECT product_name,
CEIL(price) AS Ceil_Value
FROM products;

-- MOD()
SELECT product_name,
quantity,
MOD(quantity,3) AS Remainder
FROM products;

-- ABS()
SELECT ABS(-250) AS Absolute_Value;

-- POWER()
SELECT POWER(5,3) AS Power_Result;

-- SQRT()
SELECT SQRT(144) AS Square_Root;

=========================================
MYSQL – STRING FUNCTIONS
=========================================

-- CONCAT()
SELECT CONCAT(product_name,' Quantity: ',quantity)
AS Product_Details
FROM products;

-- LENGTH()
SELECT product_name,
LENGTH(product_name) AS Name_Length
FROM products;

-- UPPER()
SELECT UPPER(product_name) AS Uppercase_Name
FROM products;

-- LOWER()
SELECT LOWER(product_name) AS Lowercase_Name
FROM products;

-- TRIM()
SELECT TRIM('    Hello MySQL    ') AS Trimmed_Text;

-- LTRIM()
SELECT LTRIM('     Hello MySQL') AS Left_Trimmed;

-- RTRIM()
SELECT RTRIM('Hello MySQL     ') AS Right_Trimmed;

-- SUBSTRING()
SELECT product_name,
SUBSTRING(product_name,1,4)
AS First_Four_Characters
FROM products;

-- REPLACE()
SELECT REPLACE(
'Welcome to MySQL',
'MySQL',
'Database'
) AS Replaced_Text;

-- REVERSE()
SELECT product_name,
REVERSE(product_name)
AS Reversed_Name
FROM products;

=========================================
MYSQL – DATE & TIME FUNCTIONS
=========================================

-- Current Date
SELECT CURDATE() AS Current_Date;

-- Current Time
SELECT CURTIME() AS Current_Time;

-- Current Date and Time
SELECT NOW() AS Current_DateTime;

-- Difference Between Dates
SELECT DATEDIFF(
'2026-12-31',
'2026-01-01'
) AS Days_Difference;
