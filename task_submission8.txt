
PYTHON – STRING MANIPULATION
============================

# Program: Working with Strings in Python

# This example demonstrates common string operations and built-in string functions.

# Defining string variables

first_name = "Karthik"
middle_initial = "R"
location = "Coimbatore"

# Combine strings using concatenation

complete_name = first_name + " " + middle_initial
print("Complete Name:", complete_name)

# Repeat a string multiple times

city_repeat = location * 3
print("Repeated City Name:", city_repeat)

# String case conversion methods

print("Convert to Uppercase:", complete_name.upper())
print("Convert to Lowercase:", complete_name.lower())
print("Convert to Title Case:", complete_name.title())
print("Capitalize First Letter:", complete_name.capitalize())

# Locate a word in a sentence

message = "Python is a versatile programming language"
print("Index of 'versatile':", message.find("versatile"))

# Count occurrences of a character

fruit = "mango"
print("Occurrences of 'a':", fruit.count("a"))

# String containing extra spaces

welcome_text = "   Welcome to Python Programming   "

# Remove unwanted spaces

print("Original Text:", welcome_text)
print("Using strip():", welcome_text.strip())
print("Using lstrip():", welcome_text.lstrip())
print("Using rstrip():", welcome_text.rstrip())

# Convert sentence into a list

line = "Python string functions are useful"
word_collection = line.split()
print("Words List:", word_collection)

# Merge list items into a single string

items = ["Python", "makes", "coding", "easy"]
combined_text = " ".join(items)
print("Combined String:", combined_text)

# Validate alphabetic characters

text_alpha = "Developer"
print("Contains Only Letters:", text_alpha.isalpha())

# Validate numeric characters

text_number = "98765"
print("Contains Only Digits:", text_number.isdigit())

# Validate alphanumeric content

text_alnum = "Code123"
print("Contains Letters and Numbers:", text_alnum.isalnum())

# Check beginning of string

greeting = "Welcome to Programming"
print("Starts with 'Welcome':", greeting.startswith("Welcome"))

# Check ending of string

print("Ends with 'Programming':", greeting.endswith("Programming"))

# Formatted string example

age = 24
profile = f"My name is {complete_name}, I am {age} years old and I live in {location}."
print("Profile Information:", profile)

# New line escape sequence

print("Example of New Line:\nPython\nDevelopment")

# Tab space escape sequence

print("Example of Tab Space:\tPython\tDevelopment")

MYSQL – SUBQUERY OPERATIONS

-- Create a database if it does not already exist
CREATE DATABASE IF NOT EXISTS company_training;

-- Select the database
USE company_training;

-- Remove employees table if available
DROP TABLE IF EXISTS employees;

-- Create employees table structure
CREATE TABLE employees (
emp_id INT PRIMARY KEY,
emp_name VARCHAR(50),
department VARCHAR(50),
salary DECIMAL(10,2)

===============================
SINGLE VALUE SUBQUERIES
=======================

-- Retrieve employee(s) earning the maximum salary
SELECT *
FROM employees
WHERE salary = (SELECT MAX(salary) FROM employees);

-- Retrieve employee(s) earning the minimum salary
SELECT *
FROM employees
WHERE salary = (SELECT MIN(salary) FROM employees);

-- Display employees whose salary exceeds the company average
SELECT *
FROM employees
WHERE salary > (SELECT AVG(salary) FROM employees);

===============================
MULTIPLE VALUE SUBQUERIES
=========================

-- IN Operator Example
SELECT *
FROM employees
WHERE department IN (
SELECT DISTINCT department
FROM employees
WHERE salary > 40000
);

-- ANY Operator Example
SELECT *
FROM employees
WHERE salary > ANY (
SELECT salary
FROM employees
WHERE department = 'Administration'
);

-- ALL Operator Example
SELECT *
FROM employees
WHERE salary > ALL (
SELECT salary
FROM employees
WHERE department = 'Quality'
);

===============================
CORRELATED SUBQUERY EXAMPLES
============================

-- Employees earning above their department average salary
SELECT e1.*
FROM employees e1
WHERE salary > (
SELECT AVG(e2.salary)
FROM employees e2
WHERE e2.department = e1.department
);

-- Employees receiving the highest salary in their department
SELECT e1.*
FROM employees e1
WHERE salary = (
SELECT MAX(e2.salary)
FROM employees e2
WHERE e2.department = e1.department
);
